\begin{frame}{I november 2025 publicerade en forskningsgrupp vid Stanford sina resultat}

\textbf{”Canaries in the Coal Mine” (Brynjolfsson et al., 2025)}

\begin{itemize}
    \item Studien analyserar amerikansk sysselsättningsstatistik efter genombrottet för generativ AI och ChatGPT
    \item Fokus ligger på så kallade AI-exponerade yrken, det vill säga yrken där arbetsuppgifterna i hög grad kan utföras eller stödjas av AI-verktyg
    \item Deras analys visar en cirka 13-procentig relativ nedgång i sysselsättning för unga i AI-exponerade yrken
    \item För äldre inom samma yrkeskategorier syns ingen motsvarande nedgång, utan snarare en positiv utveckling
\end{itemize}

\end{frame}

% DAIOE: AI EXPOSURE MEASURE
% ============================================================
\begin{frame}{Vad innebär AI-exponering?}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Dynamiskt AI-exponerings mått (DAIOE)} (Engberg et al.\ 2024)
\begin{enumerate}
  \begin{itemize}
    \item Utgår från ett antal färdigheter som AI kan utföra i varierande grad
    
    \item Matchar dessa färdigheter mot O*NET:s bedömningar av kognitiva arbetsuppgifter
    
    \item Aggregerar färdighetsmåtten till yrkesnivå
    
    \item Matchar därefter till svenska yrken med hjälp av SCB:s yrkesklassificering

    \item AI är framförallt duktig på \textit{rutinbaserade} och \textit{kognitiva} arbetsuppgifter som inte kräver sociala färdigheter
\end{itemize}
\end{enumerate}
\vspace{2mm}
\begin{itemize}
    \item Fångar vad AI teoretiskt sett \textit{kan} göra, inte faktiskt AI-adaptering
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\begin{block}{Högst exponering (Q4)}
\footnotesize
Kundtjänstpersonal, löneadministratörer, receptionists, mjukvaruutvecklare, revisorer, sekreterare
\end{block}
\vspace{2mm}
\begin{block}{Lägst exponering (Q1)}
\footnotesize
Byggarbetare, städare, sjukvårdspersonal, kockar, förskolelärare
\end{block}
\end{column}
\end{columns}

\textbf{Nyckeltal:}
\begin{itemize}
    \item 22--25 i de högst exponerade yrkena: $-5.5\%$ i början av 2025
    \item 26--30 i de högst exponerade yrkena: $-4.9\%$ i början av 2025
    \item 50+ i de högst exponerade yrkena: $+1.3\%$
\end{itemize}
\vspace{3mm}
\textbf{Färre unga nyanställs:}
\begin{itemize}
\begin{itemize}
    \item För 22–25 drivs nedgången främst av färre nyanställningar
    
    \item För 50+ ökar både nyanställningar och separationer, men nettoeffekten är positiv
\end{itemize}
\end{itemize}

\begin{frame}{Internationell jämförelse}
\vspace{3mm}
\textbf{I linje med de amerikanska resultaten:}
\begin{itemize}
    \item USA: 13\% nedgång, Sverige: 5.5\% 
    \item Samma metodologi, samma riktning
\end{itemize}
\textbf{Avviker från våra nordiska grannländer:}
\begin{itemize}
    \item Danmark: inga tydliga effekter på löner (Humlum \& Vestergaard, 2025)
    \item Finland: inga tydliga effekter på sysselsättning (Kauhanen \& Rouvinen, 2026)
    \begin{itemize}
        \item Metodmässigt lik vår egen, och resultatskillnaden kvarstår även när analysen anpassas för att efterlikna deras ansats
        
        \item Institutionella faktorer och skillnader i AI-adoption kvarstår som möjliga förklaringar
    \end{itemize}
\end{itemize}

\begin{frame}{Slutsatser}
\begin{columns}[T]

\begin{column}{0.48\textwidth}
\begin{itemize}
    \item Ingen tydlig nedgång i sysselsättningen på aggregerad nivå
    
    \item När sysselsättningen bryts ned efter ålder och yrkeskategori framträder tydliga skillnader
    
    \item Unga människor, särskilt unga kvinnor, verkar få allt svårare att etablera sig i AI-exponerade yrken
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\begin{itemize}
    \item Det återstår att se om utvecklingen förblir koncentrerad till yngre åldersgrupper eller sprider sig uppåt
    
    \item Mer forskning krävs för att identifiera mekanismerna bakom resultaten och skillnader mellan länder
    
    \item Förändringstakten är hög och bilden kan snabbt förändras igen
\end{itemize}
\end{column}
